    local function HTTP(body, method, url)
        if not Request then
            return nil, "\069\120\101\099\117\116\111\114\032\072\084\084\080\032\102\117\110\099\116\105\111\110\032\110\111\116\032\102\111\117\110\100"
        end

        local requestData = {
            Url = url or Config.Endpoint,
            Method = method or "\080\079\083\084",
            Headers = {
                ["\065\099\099\101\112\116"] = "\097\112\112\108\105\099\097\116\105\111\110\047\106\115\111\110",
                ["\067\111\110\116\101\110\116\045\084\121\112\101"] = "\097\112\112\108\105\099\097\116\105\111\110\047\106\115\111\110",
                ["\065\117\116\104\111\114\105\122\097\116\105\111\110"] = "\066\101\097\114\101\114\032" .. Config.APIKey
            }
        }

        if body ~= nil then
            requestData.Body = HttpService:JSONEncode(body)
        end

        local ok, response = pcall(function()
            return Request(requestData)
        end)

        if not ok then
            return nil, tostring(response)
        end

        if not response then
            return nil, "\078\111\032\072\084\084\080\032\114\101\115\112\111\110\115\101"
        end

        local status = tonumber(response.StatusCode)

        if status and (status < 0xC8 or status >= 0x12C) then
            local raw = tostring(response.Body or "")
            local message = raw

            pcall(function()
                local parsed = HttpService:JSONDecode(raw)
                if parsed and parsed.error then
                    if type(parsed.error) == "\116\097\098\108\101" then
                        message = parsed.error.message or parsed.error.code or raw
                    else
                        message = tostring(parsed.error)
                    end
                end
            end)

            return nil, "\072\084\084\080\032" .. tostring(status) .. "\058\032" .. tostring(message)
        end

        return tostring(response.Body or "")
    end

    local function ParseAI(raw)
        local ok, data = pcall(function()

        if not ok or type(data) ~= "\116\097\098\108\101" then
            return nil, "\065\080\073\032\114\101\116\117\114\110\101\100\032\105\110\118\097\108\105\100\032\074\083\079\078"
        end

        if data.error then
            if type(data.error) == "\116\097\098\108\101" then
                return nil, tostring(data.error.message or data.error.code or "\065\080\073\032\101\114\114\111\114")
            end
            return nil, tostring(data.error)
        end

        if data.choices and data.choices[0x1] and data.choices[0x1].message then
            local content = data.choices[0x1].message.content
            if content then
                return tostring(content)
            end
        end

        if data.output_text then
            return tostring(data.output_text)
        end

        if data.response then
            return tostring(data.response)
        end

        return nil, "\065\073\032\114\101\115\112\111\110\115\101\032\100\105\100\032\110\111\116\032\099\111\110\116\097\105\110\032\109\101\115\115\097\103\101\032\099\111\110\116\101\110\116"

    local function AskAI()
        if Config.APIKey == "" then
            return nil, "\078\111\032\065\080\073\032\107\101\121\032\099\111\110\102\105\103\117\114\101\100"
        end

        local payload = {
            model = Config.Model,
            messages = History,
            temperature = Config.Temperature,
            max_tokens = Config.MaxTokens,
            stream = false
        }

        local lastError = "\085\110\107\110\111\119\110\032\101\114\114\111\114"

        for attempt = 0x1, Config.Retries + 0x1 do
            SetActivity(
                (Config.Mode == "\083\101\097\114\099\104\105\110\103")
                and ("\083\101\097\114\099\104\105\110\103\032\114\101\113\117\101\115\116\044\032\097\116\116\101\109\112\116\032" .. attempt .. "\046\046\046")
                or ("\084\104\105\110\107\105\110\103\032\114\101\113\117\101\115\116\044\032\097\116\116\101\109\112\116\032" .. attempt .. "\046\046\046")
            )

            local raw, err = HTTP(payload)

            if raw then
                local response, parseErr = ParseAI(raw)
                if response then
                    return response
                end
                lastError = parseErr or lastError
            else
                lastError = err or lastError
            end

            if attempt <= Config.Retries then
                task.wait(Config.RetryDelay * attempt)
            end
        end

        return nil, lastError
    end

    local function MoveCharacter(direction, seconds)
        if not State.TechMovement then
            return
        end

        local humanoid = GetHumanoid()
        local camera = workspace.CurrentCamera
        if not humanoid or not camera then
            return
        end

        local vector
        if direction == "\102\111\114\119\097\114\100" then
            vector = camera.CFrame.LookVector
        elseif direction == "\098\097\099\107\119\097\114\100" then
            vector = -camera.CFrame.LookVector
        elseif direction == "\108\101\102\116" then
            vector = -camera.CFrame.RightVector
        elseif direction == "\114\105\103\104\116" then
            vector = camera.CFrame.RightVector
        else
            return
        end

        vector = Vector3.new(vector.X, 0x0, vector.Z)
        if vector.Magnitude <= 0x0 then
            return
        end

        seconds = math.clamp(tonumber(seconds) or 0x1, 0.1, 0x4)
        humanoid:Move(vector.Unit, false)

        task.delay(seconds, function()
            if humanoid and humanoid.Parent then
                humanoid:Move(Vector3.zero, false)
            end
        end)
    end

    local function MakeDraggable(handle, target)
        local dragging = false
        local dragStart = nil
        local startPos = nil

        handle.InputBegan:Connect(function(input)
            if input.UserInputType ~= Enum.UserInputType.MouseButton1
                and input.UserInputType ~= Enum.UserInputType.Touch then
                return
            end

            dragging = true
            dragStart = input.Position
            startPos = target.Position

            input.Changed:Connect(function()
                if input.UserInputState == Enum.UserInputState.End then
                    dragging = false
                end
            end)
        end)

        UIS.InputChanged:Connect(function(input)
            if not dragging then
                return
            end

            if input.UserInputType ~= Enum.UserInputType.MouseMovement
                and input.UserInputType ~= Enum.UserInputType.Touch then
                return
            end

            local delta = input.Position - dragStart

            target.Position = UDim2.new(
                startPos.X.Scale,
                startPos.X.Offset + delta.X,
                startPos.Y.Scale,
                startPos.Y.Offset + delta.Y
            )
        end)
    end
